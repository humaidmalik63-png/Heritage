#!/usr/bin/env python3
"""Validate metadata records against the schema."""
import json
import sys
from pathlib import Path
from jsonschema import validate, ValidationError

SCHEMA_PATH = Path(__file__).parent.parent / "data" / "metadata" / "schema.json"

def validate_metadata(record):
    """Validate a single metadata record."""
    with open(SCHEMA_PATH) as f:
        schema = json.load(f)
    try:
        validate(instance=record, schema=schema)
        return True, None
    except ValidationError as e:
        return False, str(e)

def main():
    if len(sys.argv) < 2:
        print("Usage: validate_metadata.py <metadata_file.json>")
        sys.exit(1)
    
    record_path = Path(sys.argv[1])
    if not record_path.exists():
        print(f"File not found: {record_path}")
        sys.exit(1)
    
    with open(record_path) as f:
        record = json.load(f)
    
    valid, error = validate_metadata(record)
    if valid:
        print(f"✓ {record_path} is valid")
    else:
        print(f"✗ {record_path} is invalid: {error}")
        sys.exit(1)

if __name__ == "__main__":
    main()

- `source`: Journal or repository
- `url`: Digital reference
- `content_type`: Article, report, photograph, etc.

1. **Ingestion**: Raw data collection and validation
2. **Preprocessing**: Image normalization, metadata extraction
3. **Annotation**: Expert labeling and classification
4. **Indexing**: Vector embedding generation
5. **Storage**: Structured database and vector index

- Privacy: No personal data in public images
- Attribution: Proper credit for all sources
- Accessibility: Open access where legally possible
- Versioning: Track dataset changes over time

"""
Baseline feature extraction interface and implementations.

This module provides a clean interface for image feature extraction
that can be easily replaced with deep learning models.
"""

from abc import ABC, abstractmethod
from typing import Union
import numpy as np
from PIL import Image
import cv2


class FeatureExtractor(ABC):
    """
    Abstract base class for image feature extractors.
    
    All feature extractors must implement this interface to ensure
    compatibility with the retrieval pipeline.
    """
    
    @abstractmethod
    def extract_features(self, image_path: str) -> np.ndarray:
        """
        Extract feature vector from an image.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Feature vector as numpy array
        """
        pass
    
    @abstractmethod
    def get_feature_dimension(self) -> int:
        """
        Get the dimensionality of the feature vector.
        
        Returns:
            Feature vector dimension
        """
        pass


class ColorHistogramExtractor(FeatureExtractor):
    """
    Color histogram feature extractor.
    
    Extracts RGB color histograms as baseline visual features.
    This is a simple, explainable method suitable for baseline comparison.
    """
    
    def __init__(self, bins_per_channel: int = 32, image_size: tuple = (128, 128)):
        """
        Initialize the color histogram extractor.
        
        Args:
            bins_per_channel: Number of bins per color channel
            image_size: Size to resize images to before feature extraction
        """
        self.bins_per_channel = bins_per_channel
        self.image_size = image_size
        self._feature_dimension = bins_per_channel * 3  # RGB
    
    def extract_features(self, image_path: str) -> np.ndarray:
        """
        Extract color histogram features from an image.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Feature vector representing color distribution
        """
        # Load and resize image
        image = Image.open(image_path).convert('RGB')
        image = image.resize(self.image_size)
        
        # Convert to numpy array
        image_array = np.array(image)
        
        # Calculate histograms for each channel
        features = []
        for channel in range(3):  # R, G, B
            channel_data = image_array[:, :, channel]
            hist, _ = np.histogram(channel_data, bins=self.bins_per_channel, range=(0, 255))
            # Normalize histogram
            hist = hist / hist.sum() if hist.sum() > 0 else hist
            features.append(hist)
        
        # Concatenate all channel features
        feature_vector = np.concatenate(features)
        
        return feature_vector.astype(np.float32)
    
    def get_feature_dimension(self) -> int:
        """Get the dimensionality of the feature vector."""
        return self._feature_dimension


class ORBFeatureExtractor(FeatureExtractor):
    """
    ORB feature extractor for baseline comparison.
    
    Uses ORB keypoints and descriptors as an alternative baseline method.
    """
    
    def __init__(self, max_features: int = 100, image_size: tuple = (128, 128)):
        """
        Initialize the ORB feature extractor.
        
        Args:
            max_features: Maximum number of ORB features to detect
            image_size: Size to resize images to before feature extraction
        """
        self.max_features = max_features
        self.image_size = image_size
        self.orb = cv2.ORB_create(maxFeatures=max_features)
        self._feature_dimension = max_features * 32  # ORB descriptors are 32 bytes
    
    def extract_features(self, image_path: str) -> np.ndarray:
        """
        Extract ORB features from an image.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Feature vector representing ORB descriptors
        """
        # Load image as grayscale
        image = cv2.imread(image_path, cv2.IMREAD_GRAYSCALE)
        if image is None:
            raise ValueError(f"Could not load image: {image_path}")
        
        # Resize image
        image = cv2.resize(image, self.image_size)
        
        # Detect keypoints and compute descriptors
        keypoints, descriptors = self.orb.detectAndCompute(image, None)
        
        if descriptors is None or len(descriptors) == 0:
            # Return zero vector if no features found
            return np.zeros(self._feature_dimension, dtype=np.float32)
        
        # Flatten and pad/truncate to fixed size
        flattened = descriptors.flatten()
        if len(flattened) < self._feature_dimension:
            # Pad with zeros
            padded = np.zeros(self._feature_dimension, dtype=np.float32)
            padded[:len(flattened)] = flattened
            return padded
        else:
            # Truncate
            return flattened[:self._feature_dimension].astype(np.float32)
    
    def get_feature_dimension(self) -> int:
        """Get the dimensionality of the feature vector."""
        return self._feature_dimension

"""
Similarity calculation interface and implementations.

This module provides methods for calculating similarity between
feature vectors, with a clean interface for future enhancements.
"""

from abc import ABC, abstractmethod
import numpy as np
from sklearn.metrics.pairwise import euclidean_distances, cosine_similarity


class SimilarityCalculator(ABC):
    """
    Abstract base class for similarity calculators.
    
    All similarity calculators must implement this interface to ensure
    compatibility with the retrieval pipeline.
    """
    
    @abstractmethod
    def calculate_similarity(self, feature1: np.ndarray, feature2: np.ndarray) -> float:
        """
        Calculate similarity between two feature vectors.
        
        Args:
            feature1: First feature vector
            feature2: Second feature vector
            
        Returns:
            Similarity score (higher means more similar)
        """
        pass
    
    @abstractmethod
    def calculate_batch_similarity(self, query_feature: np.ndarray, 
                                 database_features: np.ndarray) -> np.ndarray:
        """
        Calculate similarity between a query and a batch of database features.
        
        Args:
            query_feature: Query feature vector
            database_features: Matrix of database feature vectors
            
        Returns:
            Array of similarity scores
        """
        pass


class EuclideanSimilarity(SimilarityCalculator):
    """
    Euclidean distance-based similarity calculator.
    
    Converts Euclidean distance to similarity score (higher = more similar).
    """
    
    def calculate_similarity(self, feature1: np.ndarray, feature2: np.ndarray) -> float:
        """
        Calculate Euclidean similarity between two feature vectors.
        
        Args:
            feature1: First feature vector
            feature2: Second feature vector
            
        Returns:
            Similarity score (1 / (1 + distance))
        """
        distance = np.linalg.norm(feature1 - feature2)
        return 1.0 / (1.0 + distance)  # Convert to similarity
    
    def calculate_batch_similarity(self, query_feature: np.ndarray, 
                                 database_features: np.ndarray) -> np.ndarray:
        """
        Calculate Euclidean similarity between query and batch of features.
        
        Args:
            query_feature: Query feature vector
            database_features: Matrix of database feature vectors
            
        Returns:
            Array of similarity scores
        """
        distances = euclidean_distances([query_feature], database_features)[0]
        similarities = 1.0 / (1.0 + distances)
        return similarities


class CosineSimilarity(SimilarityCalculator):
    """
    Cosine similarity calculator.
    
    Calculates cosine similarity between feature vectors.
    """
    
    def calculate_similarity(self, feature1: np.ndarray, feature2: np.ndarray) -> float:
        """
        Calculate cosine similarity between two feature vectors.
        
        Args:
            feature1: First feature vector
            feature2: Second feature vector
            
        Returns:
            Cosine similarity score
        """
        dot_product = np.dot(feature1, feature2)
        norm1 = np.linalg.norm(feature1)
        norm2 = np.linalg.norm(feature2)
        
        if norm1 == 0 or norm2 == 0:
            return 0.0
        
        return float(dot_product / (norm1 * norm2))
    
    def calculate_batch_similarity(self, query_feature: np.ndarray, 
                                 database_features: np.ndarray) -> np.ndarray:
        """
        Calculate cosine similarity between query and batch of features.
        
        Args:
            query_feature: Query feature vector
            database_features: Matrix of database feature vectors
            
        Returns:
            Array of similarity scores
        """
        similarities = cosine_similarity([query_feature], database_features)[0]
        return similarities

"""
Baseline retrieval pipeline implementation.

This module coordinates feature extraction, similarity calculation,
and result ranking to provide a complete retrieval solution.
"""

from typing import List, Tuple, Dict, Optional
import numpy as np
from pathlib import Path

from .feature_extractor import FeatureExtractor, ColorHistogramExtractor
from .similarity_calculator import SimilarityCalculator, EuclideanSimilarity


class BaselineRetrievalPipeline:
    """
    Baseline retrieval pipeline using simple visual features.
    
    This pipeline provides a modular, reproducible baseline for visual
    retrieval that can be easily compared against deep learning approaches.
    """
    
    def __init__(self, 
                 feature_extractor: Optional[FeatureExtractor] = None,
                 similarity_calculator: Optional[SimilarityCalculator] = None):
        """
        Initialize the retrieval pipeline.
        
        Args:
            feature_extractor: Feature extractor to use (default: ColorHistogramExtractor)
            similarity_calculator: Similarity calculator to use (default: EuclideanSimilarity)
        """
        self.feature_extractor = feature_extractor or ColorHistogramExtractor()
        self.similarity_calculator = similarity_calculator or EuclideanSimilarity()
        self.database_features: Dict[str, np.ndarray] = {}
        self.database_paths: Dict[str, str] = {}
    
    def build_index(self, image_directory: str) -> None:
        """
        Build feature index from a directory of images.
        
        Args:
            image_directory: Path to directory containing images
        """
        image_dir = Path(image_directory)
        if not image_dir.exists():
            raise ValueError(f"Image directory not found: {image_directory}")
        
        # Supported image extensions
        image_extensions = {'.jpg', '.jpeg', '.png', '.bmp', '.tiff', '.webp'}
        
        # Find all image files
        image_files = []
        for ext in image_extensions:
            image_files.extend(image_dir.glob(f"*{ext}"))
            image_files.extend(image_dir.glob(f"*{ext.upper()}"))
        
        if not image_files:
            raise ValueError(f"No images found in directory: {image_directory}")
        
        # Extract features for each image
        self.database_features = {}
        self.database_paths = {}
        
        for image_path in image_files:
            try:
                features = self.feature_extractor.extract_features(str(image_path))
                image_id = image_path.stem
                self.database_features[image_id] = features
                self.database_paths[image_id] = str(image_path)
            except Exception as e:
                print(f"Warning: Could not process {image_path}: {e}")
        
        print(f"Built index with {len(self.database_features)} images")
    
    def query(self, query_image_path: str, top_k: int = 5) -> List[Tuple[str, float, str]]:
        """
        Query the index with an image and return ranked results.
        
        Args:
            query_image_path: Path to the query image
            top_k: Number of top results to return
            
        Returns:
            List of tuples (image_id, similarity_score, image_path) ranked by similarity
        """
        if not self.database_features:
            raise ValueError("Index is empty. Call build_index() first.")
        
        # Extract features from query image
        query_features = self.feature_extractor.extract_features(query_image_path)
        
        # Calculate similarities with all database images
        database_ids = list(self.database_features.keys())
        database_feature_matrix = np.array([self.database_features[id] for id in database_ids])
        
        similarities = self.similarity_calculator.calculate_batch_similarity(
            query_features, database_feature_matrix
        )
        
        # Rank results by similarity (descending)
        ranked_indices = np.argsort(similarities)[::-1]
        
        # Get top-k results
        results = []
        for i in ranked_indices[:top_k]:
            image_id = database_ids[i]
            similarity_score = float(similarities[i])
            image_path = self.database_paths[image_id]
            results.append((image_id, similarity_score, image_path))
        
        return results
    
    def get_index_size(self) -> int:
        """Get the number of images in the index."""
        return len(self.database_features)
    
    def clear_index(self) -> None:
        """Clear the current index."""
        self.database_features = {}
        self.database_paths = {}

"""
Baseline visual retrieval script.

Usage:
    python scripts/baseline_retrieval.py <query_image_path> <image_directory> [--top-k N]
"""

import argparse
import sys
from pathlib import Path

# Add project root to path
project_root = Path(__file__).parent.parent
sys.path.insert(0, str(project_root))

from ml.baseline.retrieval_pipeline import BaselineRetrievalPipeline
from ml.baseline.feature_extractor import ColorHistogramExtractor
from ml.baseline.similarity_calculator import EuclideanSimilarity


def main():
    parser = argparse.ArgumentParser(description='Baseline visual retrieval')
    parser.add_argument('query_image', help='Path to query image')
    parser.add_argument('image_directory', help='Directory containing database images')
    parser.add_argument('--top-k', type=int, default=5, 
                       help='Number of top results to return (default: 5)')
    parser.add_argument('--feature-type', choices=['color_hist', 'orb'], 
                       default='color_hist', help='Feature extraction method')
    parser.add_argument('--similarity', choices=['euclidean', 'cosine'], 
                       default='euclidean', help='Similarity metric')
    
    args = parser.parse_args()
    
    # Validate inputs
    query_path = Path(args.query_image)
    if not query_path.exists():
        print(f"Error: Query image not found: {args.query_image}")
        sys.exit(1)
    
    image_dir = Path(args.image_directory)
    if not image_dir.exists():
        print(f"Error: Image directory not found: {args.image_directory}")
        sys.exit(1)
    
    # Create feature extractor
    if args.feature_type == 'color_hist':
        feature_extractor = ColorHistogramExtractor()
    else:
        # ORB would require additional imports
        from ml.baseline.feature_extractor import ORBFeatureExtractor
        feature_extractor = ORBFeatureExtractor()
    
    # Create similarity calculator
    if args.similarity == 'euclidean':
        similarity_calculator = EuclideanSimilarity()
    else:
        from ml.baseline.similarity_calculator import CosineSimilarity
        similarity_calculator = CosineSimilarity()
    
    # Create and run pipeline
    pipeline = BaselineRetrievalPipeline(feature_extractor, similarity_calculator)
    
    print(f"Building index from: {args.image_directory}")
    pipeline.build_index(args.image_directory)
    print(f"Index size: {pipeline.get_index_size()} images")
    
    print(f"\nQuerying with: {args.query_image}")
    results = pipeline.query(str(query_path), top_k=args.top_k)
    
    print(f"\nTop {len(results)} results:")
    print("-" * 60)
    for i, (image_id, score, path) in enumerate(results, 1):
        print(f"{i}. {image_id} (Score: {score:.4f})")
        print(f"   Path: {path}")
    print("-" * 60)


if __name__ == "__main__":
    main()

# Baseline Visual Retrieval Method

## Overview
This document describes the baseline visual retrieval method implemented in Phase 3 of the project. The baseline provides a simple, explainable approach to image similarity that can be used for comparison against more advanced deep learning methods.

## Baseline Method

### Feature Representation
The baseline uses color histogram features:
- **Color Space**: RGB color space
- **Bin Configuration**: 32 bins per color channel
- **Normalization**: Histograms are normalized to sum to 1
- **Total Dimension**: 96 features (32 bins × 3 channels)

### Feature Extraction Process
1. **Image Loading**: Load image and convert to RGB format
2. **Resizing**: Resize to consistent dimensions (128×128 pixels)
3. **Histogram Calculation**: Compute histograms for each color channel
4. **Normalization**: Normalize histograms to represent probability distributions
5. **Feature Concatenation**: Combine all channel features into single vector

### Similarity Calculation
The baseline uses Euclidean distance converted to similarity:
- **Distance Metric**: Euclidean distance between feature vectors
- **Similarity Conversion**: `similarity = 1 / (1 + distance)`
- **Range**: Similarity scores range from 0 (completely different) to 1 (identical)

### Retrieval Process
1. **Index Building**: Extract features for all database images
2. **Query Processing**: Extract features from query image
3. **Similarity Computation**: Calculate similarity between query and all database images
4. **Ranking**: Sort images by similarity score (descending)
5. **Result Selection**: Return top-k most similar images

## Limitations

### Current Limitations
1. **Color Sensitivity**: Highly sensitive to lighting conditions and color variations
2. **Spatial Information**: Ignores spatial relationships between image regions
3. **Semantic Understanding**: Cannot capture semantic similarity beyond color distribution
4. **Scale Invariance**: Not invariant to scale changes
5. **Rotation Sensitivity**: Sensitive to image rotation

### Comparison with Deep Learning
The baseline will be compared against deep embedding models (CLIP, SigLIP) on:
- **Retrieval Accuracy**: Precision@k, Recall@k, mAP
- **Robustness**: Performance across different lighting, angles, and conditions
- **Semantic Understanding**: Ability to capture semantic similarity
- **Feature Dimensionality**: Compactness and efficiency of feature representation

"""
Baseline feature extraction interface and implementations.

This module provides a clean interface for image feature extraction
that can be easily replaced with deep learning models.
"""

from abc import ABC, abstractmethod
from typing import Union
import numpy as np
from PIL import Image
import cv2


class FeatureExtractor(ABC):
    """
    Abstract base class for image feature extractors.
    
    All feature extractors must implement this interface to ensure
    compatibility with the retrieval pipeline.
    """
    
    @abstractmethod
    def extract_features(self, image_path: str) -> np.ndarray:
        """
        Extract feature vector from an image.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Feature vector as numpy array
        """
        pass
    
    @abstractmethod
    def get_feature_dimension(self) -> int:
        """
        Get the dimensionality of the feature vector.
        
        Returns:
            Feature vector dimension
        """
        pass


class ColorHistogramExtractor(FeatureExtractor):
    """
    Color histogram feature extractor.
    
    Extracts RGB color histograms as baseline visual features.
    This is a simple, explainable method suitable for baseline comparison.
    """
    
    def __init__(self, bins_per_channel: int = 32, image_size: tuple = (128, 128)):
        """
        Initialize the color histogram extractor.
        
        Args:
            bins_per_channel: Number of bins per color channel
            image_size: Size to resize images to before feature extraction
        """
        self.bins_per_channel = bins_per_channel
        self.image_size = image_size
        self._feature_dimension = bins_per_channel * 3  # RGB
    
    def extract_features(self, image_path: str) -> np.ndarray:
        """
        Extract color histogram features from an image.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Feature vector representing color distribution
        """
        # Load and resize image
        image = Image.open(image_path).convert('RGB')
        image = image.resize(self.image_size)
        
        # Convert to numpy array
        image_array = np.array(image)
        
        # Calculate histograms for each channel
        features = []
        for channel in range(3):  # R, G, B
            channel_data = image_array[:, :, channel]
            hist, _ = np.histogram(channel_data, bins=self.bins_per_channel, range=(0, 255))
            # Normalize histogram
            hist = hist / hist.sum() if hist.sum() > 0 else hist
            features.append(hist)
        
        # Concatenate all channel features
        feature_vector = np.concatenate(features)
        
        return feature_vector.astype(np.float32)
    
    def get_feature_dimension(self) -> int:
        """Get the dimensionality of the feature vector."""
        return self._feature_dimension


class ORBFeatureExtractor(FeatureExtractor):
    """
    ORB feature extractor for baseline comparison.
    
    Uses ORB keypoints and descriptors as an alternative baseline method.
    """
    
    def __init__(self, max_features: int = 100, image_size: tuple = (128, 128)):
        """
        Initialize the ORB feature extractor.
        
        Args:
            max_features: Maximum number of ORB features to detect
            image_size: Size to resize images to before feature extraction
        """
        self.max_features = max_features
        self.image_size = image_size
        self.orb = cv2.ORB_create(maxFeatures=max_features)
        self._feature_dimension = max_features * 32  # ORB descriptors are 32 bytes
    
    def extract_features(self, image_path: str) -> np.ndarray:
        """
        Extract ORB features from an image.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Feature vector representing ORB descriptors
        """
        # Load image as grayscale
        image = cv2.imread(image_path, cv2.IMREAD_GRAYSCALE)
        if image is None:
            raise ValueError(f"Could not load image: {image_path}")
        
        # Resize image
        image = cv2.resize(image, self.image_size)
        
        # Detect keypoints and compute descriptors
        keypoints, descriptors = self.orb.detectAndCompute(image, None)
        
        if descriptors is None or len(descriptors) == 0:
            # Return zero vector if no features found
            return np.zeros(self._feature_dimension, dtype=np.float32)
        
        # Flatten and pad/truncate to fixed size
        flattened = descriptors.flatten()
        if len(flattened) < self._feature_dimension:
            # Pad with zeros
            padded = np.zeros(self._feature_dimension, dtype=np.float32)
            padded[:len(flattened)] = flattened
            return padded
        else:
            # Truncate
            return flattened[:self._feature_dimension].astype(np.float32)
    
    def get_feature_dimension(self) -> int:
        """Get the dimensionality of the feature vector."""
        return self._feature_dimension

"""
Similarity calculation interface and implementations.

This module provides methods for calculating similarity between
feature vectors, with a clean interface for future enhancements.
"""

from abc import ABC, abstractmethod
import numpy as np
from sklearn.metrics.pairwise import euclidean_distances, cosine_similarity


class SimilarityCalculator(ABC):
    """
    Abstract base class for similarity calculators.
    
    All similarity calculators must implement this interface to ensure
    compatibility with the retrieval pipeline.
    """
    
    @abstractmethod
    def calculate_similarity(self, feature1: np.ndarray, feature2: np.ndarray) -> float:
        """
        Calculate similarity between two feature vectors.
        
        Args:
            feature1: First feature vector
            feature2: Second feature vector
            
        Returns:
            Similarity score (higher means more similar)
        """
        pass
    
    @abstractmethod
    def calculate_batch_similarity(self, query_feature: np.ndarray, 
                                 database_features: np.ndarray) -> np.ndarray:
        """
        Calculate similarity between a query and a batch of database features.
        
        Args:
            query_feature: Query feature vector
            database_features: Matrix of database feature vectors
            
        Returns:
            Array of similarity scores
        """
        pass


class EuclideanSimilarity(SimilarityCalculator):
    """
    Euclidean distance-based similarity calculator.
    
    Converts Euclidean distance to similarity score (higher = more similar).
    """
    
    def calculate_similarity(self, feature1: np.ndarray, feature2: np.ndarray) -> float:
        """
        Calculate Euclidean similarity between two feature vectors.
        
        Args:
            feature1: First feature vector
            feature2: Second feature vector
            
        Returns:
            Similarity score (1 / (1 + distance))
        """
        distance = np.linalg.norm(feature1 - feature2)
        return 1.0 / (1.0 + distance)  # Convert to similarity
    
    def calculate_batch_similarity(self, query_feature: np.ndarray, 
                                 database_features: np.ndarray) -> np.ndarray:
        """
        Calculate Euclidean similarity between query and batch of features.
        
        Args:
            query_feature: Query feature vector
            database_features: Matrix of database feature vectors
            
        Returns:
            Array of similarity scores
        """
        distances = euclidean_distances([query_feature], database_features)[0]
        similarities = 1.0 / (1.0 + distances)
        return similarities


class CosineSimilarity(SimilarityCalculator):
    """
    Cosine similarity calculator.
    
    Calculates cosine similarity between feature vectors.
    """
    
    def calculate_similarity(self, feature1: np.ndarray, feature2: np.ndarray) -> float:
        """
        Calculate cosine similarity between two feature vectors.
        
        Args:
            feature1: First feature vector
            feature2: Second feature vector
            
        Returns:
            Cosine similarity score
        """
        dot_product = np.dot(feature1, feature2)
        norm1 = np.linalg.norm(feature1)
        norm2 = np.linalg.norm(feature2)
        
        if norm1 == 0 or norm2 == 0:
            return 0.0
        
        return float(dot_product / (norm1 * norm2))
    
    def calculate_batch_similarity(self, query_feature: np.ndarray, 
                                 database_features: np.ndarray) -> np.ndarray:
        """
        Calculate cosine similarity between query and batch of features.
        
        Args:
            query_feature: Query feature vector
            database_features: Matrix of database feature vectors
            
        Returns:
            Array of similarity scores
        """
        similarities = cosine_similarity([query_feature], database_features)[0]
        return similarities

"""
Baseline retrieval pipeline implementation.

This module coordinates feature extraction, similarity calculation,
and result ranking to provide a complete retrieval solution.
"""

from typing import List, Tuple, Dict, Optional
import numpy as np
from pathlib import Path

from .feature_extractor import FeatureExtractor, ColorHistogramExtractor
from .similarity_calculator import SimilarityCalculator, EuclideanSimilarity


class BaselineRetrievalPipeline:
    """
    Baseline retrieval pipeline using simple visual features.
    
    This pipeline provides a modular, reproducible baseline for visual
    retrieval that can be easily compared against deep learning approaches.
    """
    
    def __init__(self, 
                 feature_extractor: Optional[FeatureExtractor] = None,
                 similarity_calculator: Optional[SimilarityCalculator] = None):
        """
        Initialize the retrieval pipeline.
        
        Args:
            feature_extractor: Feature extractor to use (default: ColorHistogramExtractor)
            similarity_calculator: Similarity calculator to use (default: EuclideanSimilarity)
        """
        self.feature_extractor = feature_extractor or ColorHistogramExtractor()
        self.similarity_calculator = similarity_calculator or EuclideanSimilarity()
        self.database_features: Dict[str, np.ndarray] = {}
        self.database_paths: Dict[str, str] = {}
    
    def build_index(self, image_directory: str) -> None:
        """
        Build feature index from a directory of images.
        
        Args:
            image_directory: Path to directory containing images
        """
        image_dir = Path(image_directory)
        if not image_dir.exists():
            raise ValueError(f"Image directory not found: {image_directory}")
        
        # Supported image extensions
        image_extensions = {'.jpg', '.jpeg', '.png', '.bmp', '.tiff', '.webp'}
        
        # Find all image files
        image_files = []
        for ext in image_extensions:
            image_files.extend(image_dir.glob(f"*{ext}"))
            image_files.extend(image_dir.glob(f"*{ext.upper()}"))
        
        if not image_files:
            raise ValueError(f"No images found in directory: {image_directory}")
        
        # Extract features for each image
        self.database_features = {}
        self.database_paths = {}
        
        for image_path in image_files:
            try:
                features = self.feature_extractor.extract_features(str(image_path))
                image_id = image_path.stem
                self.database_features[image_id] = features
                self.database_paths[image_id] = str(image_path)
            except Exception as e:
                print(f"Warning: Could not process {image_path}: {e}")
        
        print(f"Built index with {len(self.database_features)} images")
    
    def query(self, query_image_path: str, top_k: int = 5) -> List[Tuple[str, float, str]]:
        """
        Query the index with an image and return ranked results.
        
        Args:
            query_image_path: Path to the query image
            top_k: Number of top results to return
            
        Returns:
            List of tuples (image_id, similarity_score, image_path) ranked by similarity
        """
        if not self.database_features:
            raise ValueError("Index is empty. Call build_index() first.")
        
        # Extract features from query image
        query_features = self.feature_extractor.extract_features(query_image_path)
        
        # Calculate similarities with all database images
        database_ids = list(self.database_features.keys())
        database_feature_matrix = np.array([self.database_features[id] for id in database_ids])
        
        similarities = self.similarity_calculator.calculate_batch_similarity(
            query_features, database_feature_matrix
        )
        
        # Rank results by similarity (descending)
        ranked_indices = np.argsort(similarities)[::-1]
        
        # Get top-k results
        results = []
        for i in ranked_indices[:top_k]:
            image_id = database_ids[i]
            similarity_score = float(similarities[i])
            image_path = self.database_paths[image_id]
            results.append((image_id, similarity_score, image_path))
        
        return results
    
    def get_index_size(self) -> int:
        """Get the number of images in the index."""
        return len(self.database_features)
    
    def clear_index(self) -> None:
        """Clear the current index."""
        self.database_features = {}
        self.database_paths = {}

"""
Baseline visual retrieval script.

Usage:
    python scripts/baseline_retrieval.py <query_image_path> <image_directory> [--top-k N]
"""

import argparse
import sys
from pathlib import Path

# Add project root to path
project_root = Path(__file__).parent.parent
sys.path.insert(0, str(project_root))

from ml.baseline.retrieval_pipeline import BaselineRetrievalPipeline
from ml.baseline.feature_extractor import ColorHistogramExtractor
from ml.baseline.similarity_calculator import EuclideanSimilarity


def main():
    parser = argparse.ArgumentParser(description='Baseline visual retrieval')
    parser.add_argument('query_image', help='Path to query image')
    parser.add_argument('image_directory', help='Directory containing database images')
    parser.add_argument('--top-k', type=int, default=5, 
                       help='Number of top results to return (default: 5)')
    parser.add_argument('--feature-type', choices=['color_hist', 'orb'], 
                       default='color_hist', help='Feature extraction method')
    parser.add_argument('--similarity', choices=['euclidean', 'cosine'], 
                       default='euclidean', help='Similarity metric')
    
    args = parser.parse_args()
    
    # Validate inputs
    query_path = Path(args.query_image)
    if not query_path.exists():
        print(f"Error: Query image not found: {args.query_image}")
        sys.exit(1)
    
    image_dir = Path(args.image_directory)
    if not image_dir.exists():
        print(f"Error: Image directory not found: {args.image_directory}")
        sys.exit(1)
    
    # Create feature extractor
    if args.feature_type == 'color_hist':
        feature_extractor = ColorHistogramExtractor()
    else:
        # ORB would require additional imports
        from ml.baseline.feature_extractor import ORBFeatureExtractor
        feature_extractor = ORBFeatureExtractor()
    
    # Create similarity calculator
    if args.similarity == 'euclidean':
        similarity_calculator = EuclideanSimilarity()
    else:
        from ml.baseline.similarity_calculator import CosineSimilarity
        similarity_calculator = CosineSimilarity()
    
    # Create and run pipeline
    pipeline = BaselineRetrievalPipeline(feature_extractor, similarity_calculator)
    
    print(f"Building index from: {args.image_directory}")
    pipeline.build_index(args.image_directory)
    print(f"Index size: {pipeline.get_index_size()} images")
    
    print(f"\nQuerying with: {args.query_image}")
    results = pipeline.query(str(query_path), top_k=args.top_k)
    
    print(f"\nTop {len(results)} results:")
    print("-" * 60)
    for i, (image_id, score, path) in enumerate(results, 1):
        print(f"{i}. {image_id} (Score: {score:.4f})")
        print(f"   Path: {path}")
    print("-" * 60)


if __name__ == "__main__":
    main()

This document describes the baseline visual retrieval method implemented in Phase 3 of the project. The baseline provides a simple, explainable approach to image similarity that can be used for comparison against more advanced deep learning methods.

## Baseline Method

### Feature Representation
The baseline uses color histogram features:
- **Color Space**: RGB color space
- **Bin Configuration**: 32 bins per color channel
- **Normalization**: Histograms are normalized to sum to 1
- **Total Dimension**: 96 features (32 bins × 3 channels)

### Feature Extraction Process
1. **Image Loading**: Load image and convert to RGB format
2. **Resizing**: Resize to consistent dimensions (128×128 pixels)
3. **Histogram Calculation**: Compute histograms for each color channel
4. **Normalization**: Normalize histograms to represent probability distributions
5. **Feature Concatenation**: Combine all channel features into single vector

### Similarity Calculation
The baseline uses Euclidean distance converted to similarity:
- **Distance Metric**: Euclidean distance between feature vectors
- **Similarity Conversion**: `similarity = 1 / (1 + distance)`
- **Range**: Similarity scores range from 0 (completely different) to 1 (identical)

### Retrieval Process
1. **Index Building**: Extract features for all database images
2. **Query Processing**: Extract features from query image
3. **Similarity Computation**: Calculate similarity between query and all database images
4. **Ranking**: Sort images by similarity score (descending)
5. **Result Selection**: Return top-k most similar images

## Limitations

### Current Limitations
1. **Color Sensitivity**: Highly sensitive to lighting conditions and color variations
2. **Spatial Information**: Ignores spatial relationships between image regions
3. **Semantic Understanding**: Cannot capture semantic similarity beyond color distribution
4. **Scale Invariance**: Not invariant to scale changes
5. **Rotation Sensitivity**: Sensitive to image rotation

### Comparison with Deep Learning
The baseline will be compared against deep embedding models (CLIP, SigLIP) on:
- **Retrieval Accuracy**: Precision@k, Recall@k, mAP
- **Robustness**: Performance across different lighting, angles, and conditions
- **Semantic Understanding**: Ability to capture semantic similarity

"""
Baseline feature extraction interface and implementations.

This module provides a clean interface for image feature extraction
that can be easily replaced with deep learning models.
"""

from abc import ABC, abstractmethod
from typing import Union
import numpy as np
from PIL import Image
import cv2


class FeatureExtractor(ABC):
    """
    Abstract base class for image feature extractors.
    
    All feature extractors must implement this interface to ensure
    compatibility with the retrieval pipeline.
    """
    
    @abstractmethod
    def extract_features(self, image_path: str) -> np.ndarray:
        """
        Extract feature vector from an image.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Feature vector as numpy array
        """
        pass
    
    @abstractmethod
    def get_feature_dimension(self) -> int:
        """
        Get the dimensionality of the feature vector.
        
        Returns:
            Feature vector dimension
        """
        pass


class ColorHistogramExtractor(FeatureExtractor):
    """
    Color histogram feature extractor.
    
    Extracts RGB color histograms as baseline visual features.
    This is a simple, explainable method suitable for baseline comparison.
    """
    
    def __init__(self, bins_per_channel: int = 32, image_size: tuple = (128, 128)):
        """
        Initialize the color histogram extractor.
        
        Args:
            bins_per_channel: Number of bins per color channel
            image_size: Size to resize images to before feature extraction
        """
        self.bins_per_channel = bins_per_channel
        self.image_size = image_size
        self._feature_dimension = bins_per_channel * 3  # RGB
    
    def extract_features(self, image_path: str) -> np.ndarray:
        """
        Extract color histogram features from an image.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Feature vector representing color distribution
        """
        # Load and resize image
        image = Image.open(image_path).convert('RGB')
        image = image.resize(self.image_size)
        
        # Convert to numpy array
        image_array = np.array(image)
        
        # Calculate histograms for each channel
        features = []
        for channel in range(3):  # R, G, B
            channel_data = image_array[:, :, channel]
            hist, _ = np.histogram(channel_data, bins=self.bins_per_channel, range=(0, 255))
            # Normalize histogram
            hist = hist / hist.sum() if hist.sum() > 0 else hist
            features.append(hist)
        
        # Concatenate all channel features
        feature_vector = np.concatenate(features)
        
        return feature_vector.astype(np.float32)
    
    def get_feature_dimension(self) -> int:
        """Get the dimensionality of the feature vector."""
        return self._feature_dimension


class ORBFeatureExtractor(FeatureExtractor):
    """
    ORB feature extractor for baseline comparison.
    
    Uses ORB keypoints and descriptors as an alternative baseline method.
    """
    
    def __init__(self, max_features: int = 100, image_size: tuple = (128, 128)):
        """
        Initialize the ORB feature extractor.
        
        Args:
            max_features: Maximum number of ORB features to detect
            image_size: Size to resize images to before feature extraction
        """
        self.max_features = max_features
        self.image_size = image_size
        self.orb = cv2.ORB_create(maxFeatures=max_features)
        self._feature_dimension = max_features * 32  # ORB descriptors are 32 bytes
    
    def extract_features(self, image_path: str) -> np.ndarray:
        """
        Extract ORB features from an image.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Feature vector representing ORB descriptors
        """
        # Load image as grayscale
        image = cv2.imread(image_path, cv2.IMREAD_GRAYSCALE)
        if image is None:
            raise ValueError(f"Could not load image: {image_path}")
        
        # Resize image
        image = cv2.resize(image, self.image_size)
        
        # Detect keypoints and compute descriptors
        keypoints, descriptors = self.orb.detectAndCompute(image, None)
        
        if descriptors is None or len(descriptors) == 0:
            # Return zero vector if no features found
            return np.zeros(self._feature_dimension, dtype=np.float32)
        
        # Flatten and pad/truncate to fixed size
        flattened = descriptors.flatten()
        if len(flattened) < self._feature_dimension:
            # Pad with zeros
            padded = np.zeros(self._feature_dimension, dtype=np.float32)
            padded[:len(flattened)] = flattened
            return padded
        else:
            # Truncate
            return flattened[:self._feature_dimension].astype(np.float32)
    
    def get_feature_dimension(self) -> int:
        """Get the dimensionality of the feature vector."""
        return self._feature_dimension

"""
Similarity calculation interface and implementations.

This module provides methods for calculating similarity between
feature vectors, with a clean interface for future enhancements.
"""

from abc import ABC, abstractmethod
import numpy as np
from sklearn.metrics.pairwise import euclidean_distances, cosine_similarity


class SimilarityCalculator(ABC):
    """
    Abstract base class for similarity calculators.
    
    All similarity calculators must implement this interface to ensure
    compatibility with the retrieval pipeline.
    """
    
    @abstractmethod
    def calculate_similarity(self, feature1: np.ndarray, feature2: np.ndarray) -> float:
        """
        Calculate similarity between two feature vectors.
        
        Args:
            feature1: First feature vector
            feature2: Second feature vector
            
        Returns:
            Similarity score (higher means more similar)
        """
        pass
    
    @abstractmethod
    def calculate_batch_similarity(self, query_feature: np.ndarray, 
                                 database_features: np.ndarray) -> np.ndarray:
        """
        Calculate similarity between a query and a batch of database features.
        
        Args:
            query_feature: Query feature vector
            database_features: Matrix of database feature vectors
            
        Returns:
            Array of similarity scores
        """
        pass


class EuclideanSimilarity(SimilarityCalculator):
    """
    Euclidean distance-based similarity calculator.
    
    Converts Euclidean distance to similarity score (higher = more similar).
    """
    
    def calculate_similarity(self, feature1: np.ndarray, feature2: np.ndarray) -> float:
        """
        Calculate Euclidean similarity between two feature vectors.
        
        Args:
            feature1: First feature vector
            feature2: Second feature vector
            
        Returns:
            Similarity score (1 / (1 + distance))
        """
        distance = np.linalg.norm(feature1 - feature2)
        return 1.0 / (1.0 + distance)  # Convert to similarity
    
    def calculate_batch_similarity(self, query_feature: np.ndarray, 
                                 database_features: np.ndarray) -> np.ndarray:
        """
        Calculate Euclidean similarity between query and batch of features.
        
        Args:
            query_feature: Query feature vector
            database_features: Matrix of database feature vectors
            
        Returns:
            Array of similarity scores
        """
        distances = euclidean_distances([query_feature], database_features)[0]
        similarities = 1.0 / (1.0 + distances)
        return similarities


class CosineSimilarity(SimilarityCalculator):
    """
    Cosine similarity calculator.
    
    Calculates cosine similarity between feature vectors.
    """
    
    def calculate_similarity(self, feature1: np.ndarray, feature2: np.ndarray) -> float:
        """
        Calculate cosine similarity between two feature vectors.
        
        Args:
            feature1: First feature vector
            feature2: Second feature vector
            
        Returns:
            Cosine similarity score
        """
        dot_product = np.dot(feature1, feature2)
        norm1 = np.linalg.norm(feature1)
        norm2 = np.linalg.norm(feature2)
        
        if norm1 == 0 or norm2 == 0:
            return 0.0
        
        return float(dot_product / (norm1 * norm2))
    
    def calculate_batch_similarity(self, query_feature: np.ndarray, 
                                 database_features: np.ndarray) -> np.ndarray:
        """
        Calculate cosine similarity between query and batch of features.
        
        Args:
            query_feature: Query feature vector
            database_features: Matrix of database feature vectors
            
        Returns:
            Array of similarity scores
        """
        similarities = cosine_similarity([query_feature], database_features)[0]
        return similarities

"""
Baseline retrieval pipeline implementation.

This module coordinates feature extraction, similarity calculation,
and result ranking to provide a complete retrieval solution.
"""

from typing import List, Tuple, Dict, Optional
import numpy as np
from pathlib import Path

from .feature_extractor import FeatureExtractor, ColorHistogramExtractor
from .similarity_calculator import SimilarityCalculator, EuclideanSimilarity


class BaselineRetrievalPipeline:
    """
    Baseline retrieval pipeline using simple visual features.
    
    This pipeline provides a modular, reproducible baseline for visual
    retrieval that can be easily compared against deep learning approaches.
    """
    
    def __init__(self, 
                 feature_extractor: Optional[FeatureExtractor] = None,
                 similarity_calculator: Optional[SimilarityCalculator] = None):
        """
        Initialize the retrieval pipeline.
        
        Args:
            feature_extractor: Feature extractor to use (default: ColorHistogramExtractor)
            similarity_calculator: Similarity calculator to use (default: EuclideanSimilarity)
        """
        self.feature_extractor = feature_extractor or ColorHistogramExtractor()
        self.similarity_calculator = similarity_calculator or EuclideanSimilarity()
        self.database_features: Dict[str, np.ndarray] = {}
        self.database_paths: Dict[str, str] = {}
    
    def build_index(self, image_directory: str) -> None:
        """
        Build feature index from a directory of images.
        
        Args:
            image_directory: Path to directory containing images
        """
        image_dir = Path(image_directory)
        if not image_dir.exists():
            raise ValueError(f"Image directory not found: {image_directory}")
        
        # Supported image extensions
        image_extensions = {'.jpg', '.jpeg', '.png', '.bmp', '.tiff', '.webp'}
        
        # Find all image files
        image_files = []
        for ext in image_extensions:
            image_files.extend(image_dir.glob(f"*{ext}"))
            image_files.extend(image_dir.glob(f"*{ext.upper()}"))
        
        if not image_files:
            raise ValueError(f"No images found in directory: {image_directory}")
        
        # Extract features for each image
        self.database_features = {}
        self.database_paths = {}
        
        for image_path in image_files:
            try:
                features = self.feature_extractor.extract_features(str(image_path))
                image_id = image_path.stem
                self.database_features[image_id] = features
                self.database_paths[image_id] = str(image_path)
            except Exception as e:
                print(f"Warning: Could not process {image_path}: {e}")
        
        print(f"Built index with {len(self.database_features)} images")
    
    def query(self, query_image_path: str, top_k: int = 5) -> List[Tuple[str, float, str]]:
        """
        Query the index with an image and return ranked results.
        
        Args:
            query_image_path: Path to the query image
            top_k: Number of top results to return
            
        Returns:
            List of tuples (image_id, similarity_score, image_path) ranked by similarity
        """
        if not self.database_features:
            raise ValueError("Index is empty. Call build_index() first.")
        
        # Extract features from query image
        query_features = self.feature_extractor.extract_features(query_image_path)
        
        # Calculate similarities with all database images
        database_ids = list(self.database_features.keys())
        database_feature_matrix = np.array([self.database_features[id] for id in database_ids])
        
        similarities = self.similarity_calculator.calculate_batch_similarity(
            query_features, database_feature_matrix
        )
        
        # Rank results by similarity (descending)
        ranked_indices = np.argsort(similarities)[::-1]
        
        # Get top-k results
        results = []
        for i in ranked_indices[:top_k]:
            image_id = database_ids[i]
            similarity_score = float(similarities[i])
            image_path = self.database_paths[image_id]
            results.append((image_id, similarity_score, image_path))
        
        return results
    
    def get_index_size(self) -> int:
        """Get the number of images in the index."""
        return len(self.database_features)
    
    def clear_index(self) -> None:
        """Clear the current index."""
        self.database_features = {}
        self.database_paths = {}

# Baseline Visual Retrieval Method

## Overview
This document describes the baseline visual retrieval method implemented in Phase 3 of the project. The baseline provides a simple, explainable approach to image similarity that can be used for comparison against more advanced deep learning methods.

## Baseline Method

### Feature Representation
The baseline uses color histogram features:
- **Color Space**: RGB color space
- **Bin Configuration**: 32 bins per color channel
- **Normalization**: Histograms are normalized to sum to 1
- **Total Dimension**: 96 features (32 bins × 3 channels)

### Feature Extraction Process
1. **Image Loading**: Load image and convert to RGB format
2. **Resizing**: Resize to consistent dimensions (128×128 pixels)
3. **Histogram Calculation**: Compute histograms for each color channel
4. **Normalization**: Normalize histograms to represent probability distributions
5. **Feature Concatenation**: Combine all channel features into single vector

### Similarity Calculation
The baseline uses Euclidean distance converted to similarity:
- **Distance Metric**: Euclidean distance between feature vectors
- **Similarity Conversion**: `similarity = 1 / (1 + distance)`
- **Range**: Similarity scores range from 0 (completely different) to 1 (identical)

### Retrieval Process
1. **Index Building**: Extract features for all database images
2. **Query Processing**: Extract features from query image
3. **Similarity Computation**: Calculate similarity between query and all database images
4. **Ranking**: Sort images by similarity score (descending)
5. **Result Selection**: Return top-k most similar images

## Limitations

### Current Limitations
1. **Color Sensitivity**: Highly sensitive to lighting conditions and color variations
2. **Spatial Information**: Ignores spatial relationships between image regions
3. **Semantic Understanding**: Cannot capture semantic similarity beyond color distribution
4. **Scale Invariance**: Not invariant to scale changes
5. **Rotation Sensitivity**: Sensitive to image rotation

### Comparison with Deep Learning
The baseline will be compared against deep embedding models (CLIP, SigLIP) on:
- **Retrieval Accuracy**: Precision@k, Recall@k, mAP
- **Robustness**: Performance across different lighting, angles, and conditions
- **Semantic Understanding**: Ability to capture semantic similarity
- **Feature Dimensionality**: Compactness and efficiency of feature representation

The baseline implements clear interfaces that allow for easy replacement:

"""
Baseline feature extraction interface and implementations.

This module provides a clean interface for image feature extraction
that can be easily replaced with deep learning models.
"""

from abc import ABC, abstractmethod
from typing import Union
import numpy as np
from PIL import Image
import cv2


class FeatureExtractor(ABC):
    """
    Abstract base class for image feature extractors.
    
    All feature extractors must implement this interface to ensure
    compatibility with the retrieval pipeline.
    """
    
    @abstractmethod
    def extract_features(self, image_path: str) -> np.ndarray:
        """
        Extract feature vector from an image.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Feature vector as numpy array
        """
        pass
    
    @abstractmethod
    def get_feature_dimension(self) -> int:
        """
        Get the dimensionality of the feature vector.
        
        Returns:
            Feature vector dimension
        """
        pass


class ColorHistogramExtractor(FeatureExtractor):
    """
    Color histogram feature extractor.
    
    Extracts RGB color histograms as baseline visual features.
    This is a simple, explainable method suitable for baseline comparison.
    """
    
    def __init__(self, bins_per_channel: int = 32, image_size: tuple = (128, 128)):
        """
        Initialize the color histogram extractor.
        
        Args:
            bins_per_channel: Number of bins per color channel
            image_size: Size to resize images to before feature extraction
        """
        self.bins_per_channel = bins_per_channel
        self.image_size = image_size
        self._feature_dimension = bins_per_channel * 3  # RGB
    
    def extract_features(self, image_path: str) -> np.ndarray:
        """
        Extract color histogram features from an image.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Feature vector representing color distribution
        """
        # Load and resize image
        image = Image.open(image_path).convert('RGB')
        image = image.resize(self.image_size)
        
        # Convert to numpy array
        image_array = np.array(image)
        
        # Calculate histograms for each channel
        features = []
        for channel in range(3):  # R, G, B
            channel_data = image_array[:, :, channel]
            hist, _ = np.histogram(channel_data, bins=self.bins_per_channel, range=(0, 255))
            # Normalize histogram
            hist = hist / hist.sum() if hist.sum() > 0 else hist
            features.append(hist)
        
        # Concatenate all channel features
        feature_vector = np.concatenate(features)
        
        return feature_vector.astype(np.float32)
    
    def get_feature_dimension(self) -> int:
        """Get the dimensionality of the feature vector."""
        return self._feature_dimension


class ORBFeatureExtractor(FeatureExtractor):
    """
    ORB feature extractor for baseline comparison.
    
    Uses ORB keypoints and descriptors as an alternative baseline method.
    """
    
    def __init__(self, max_features: int = 100, image_size: tuple = (128, 128)):
        """
        Initialize the ORB feature extractor.
        
        Args:
            max_features: Maximum number of ORB features to detect
            image_size: Size to resize images to before feature extraction
        """
        self.max_features = max_features
        self.image_size = image_size
        self.orb = cv2.ORB_create(maxFeatures=max_features)
        self._feature_dimension = max_features * 32  # ORB descriptors are 32 bytes
    
    def extract_features(self, image_path: str) -> np.ndarray:
        """
        Extract ORB features from an image.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Feature vector representing ORB descriptors
        """
        # Load image as grayscale
        image = cv2.imread(image_path, cv2.IMREAD_GRAYSCALE)
        if image is None:
            raise ValueError(f"Could not load image: {image_path}")
        
        # Resize image
        image = cv2.resize(image, self.image_size)
        
        # Detect keypoints and compute descriptors
        keypoints, descriptors = self.orb.detectAndCompute(image, None)
        
        if descriptors is None or len(descriptors) == 0:
            # Return zero vector if no features found
            return np.zeros(self._feature_dimension, dtype=np.float32)
        
        # Flatten and pad/truncate to fixed size
        flattened = descriptors.flatten()
        if len(flattened) < self._feature_dimension:
            # Pad with zeros
            padded = np.zeros(self._feature_dimension, dtype=np.float32)
            padded[:len(flattened)] = flattened
            return padded
        else:
            # Truncate
            return flattened[:self._feature_dimension].astype(np.float32)
    
    def get_feature_dimension(self) -> int:
        """Get the dimensionality of the feature vector."""
        return self._feature_dimension

"""
Similarity calculation interface and implementations.

This module provides methods for calculating similarity between
feature vectors, with a clean interface for future enhancements.
"""

from abc import ABC, abstractmethod
import numpy as np
from sklearn.metrics.pairwise import euclidean_distances, cosine_similarity


class SimilarityCalculator(ABC):
    """
    Abstract base class for similarity calculators.
    
    All similarity calculators must implement this interface to ensure
    compatibility with the retrieval pipeline.
    """
    
    @abstractmethod
    def calculate_similarity(self, feature1: np.ndarray, feature2: np.ndarray) -> float:
        """
        Calculate similarity between two feature vectors.
        
        Args:
            feature1: First feature vector
            feature2: Second feature vector
            
        Returns:
            Similarity score (higher means more similar)
        """
        pass
    
    @abstractmethod
    def calculate_batch_similarity(self, query_feature: np.ndarray, 
                                 database_features: np.ndarray) -> np.ndarray:
        """
        Calculate similarity between a query and a batch of database features.
        
        Args:
            query_feature: Query feature vector
            database_features: Matrix of database feature vectors
            
        Returns:
            Array of similarity scores
        """
        pass


class EuclideanSimilarity(SimilarityCalculator):
    """
    Euclidean distance-based similarity calculator.
    
    Converts Euclidean distance to similarity score (higher = more similar).
    """
    
    def calculate_similarity(self, feature1: np.ndarray, feature2: np.ndarray) -> float:
        """
        Calculate Euclidean similarity between two feature vectors.
        
        Args:
            feature1: First feature vector
            feature2: Second feature vector
            
        Returns:
            Similarity score (1 / (1 + distance))
        """
        distance = np.linalg.norm(feature1 - feature2)
        return 1.0 / (1.0 + distance)  # Convert to similarity
    
    def calculate_batch_similarity(self, query_feature: np.ndarray, 
                                 database_features: np.ndarray) -> np.ndarray:
        """
        Calculate Euclidean similarity between query and batch of features.
        
        Args:
            query_feature: Query feature vector
            database_features: Matrix of database feature vectors
            
        Returns:
            Array of similarity scores
        """
        distances = euclidean_distances([query_feature], database_features)[0]
        similarities = 1.0 / (1.0 + distances)
        return similarities


class CosineSimilarity(SimilarityCalculator):
    """
    Cosine similarity calculator.
    
    Calculates cosine similarity between feature vectors.
    """
    
    def calculate_similarity(self, feature1: np.ndarray, feature2: np.ndarray) -> float:
        """
        Calculate cosine similarity between two feature vectors.
        
        Args:
            feature1: First feature vector
            feature2: Second feature vector
            
        Returns:
            Cosine similarity score
        """
        dot_product = np.dot(feature1, feature2)
        norm1 = np.linalg.norm(feature1)
        norm2 = np.linalg.norm(feature2)
        
        if norm1 == 0 or norm2 == 0:
            return 0.0
        
        return float(dot_product / (norm1 * norm2))
    
    def calculate_batch_similarity(self, query_feature: np.ndarray, 
                                 database_features: np.ndarray) -> np.ndarray:
        """
        Calculate cosine similarity between query and batch of features.
        
        Args:
            query_feature: Query feature vector
            database_features: Matrix of database feature vectors
            
        Returns:
            Array of similarity scores
        """
        similarities = cosine_similarity([query_feature], database_features)[0]
        return similarities

"""
Baseline retrieval pipeline implementation.

This module coordinates feature extraction, similarity calculation,
and result ranking to provide a complete retrieval solution.
"""

from typing import List, Tuple, Dict, Optional
import numpy as np
from pathlib import Path

from .feature_extractor import FeatureExtractor, ColorHistogramExtractor
from .similarity_calculator import SimilarityCalculator, EuclideanSimilarity


class BaselineRetrievalPipeline:
    """
    Baseline retrieval pipeline using simple visual features.
    
    This pipeline provides a modular, reproducible baseline for visual
    retrieval that can be easily compared against deep learning approaches.
    """
    
    def __init__(self, 
                 feature_extractor: Optional[FeatureExtractor] = None,
                 similarity_calculator: Optional[SimilarityCalculator] = None):
        """
        Initialize the retrieval pipeline.
        
        Args:
            feature_extractor: Feature extractor to use (default: ColorHistogramExtractor)
            similarity_calculator: Similarity calculator to use (default: EuclideanSimilarity)
        """
        self.feature_extractor = feature_extractor or ColorHistogramExtractor()
        self.similarity_calculator = similarity_calculator or EuclideanSimilarity()
        self.database_features: Dict[str, np.ndarray] = {}
        self.database_paths: Dict[str, str] = {}
    
    def build_index(self, image_directory: str) -> None:
        """
        Build feature index from a directory of images.
        
        Args:
            image_directory: Path to directory containing images
        """
        image_dir = Path(image_directory)
        if not image_dir.exists():
            raise ValueError(f"Image directory not found: {image_directory}")
        
        # Supported image extensions
        image_extensions = {'.jpg', '.jpeg', '.png', '.bmp', '.tiff', '.webp'}
        
        # Find all image files
        image_files = []
        for ext in image_extensions:
            image_files.extend(image_dir.glob(f"*{ext}"))
            image_files.extend(image_dir.glob(f"*{ext.upper()}"))
        
        if not image_files:
            raise ValueError(f"No images found in directory: {image_directory}")
        
        # Extract features for each image
        self.database_features = {}
        self.database_paths = {}
        
        for image_path in image_files:
            try:
                features = self.feature_extractor.extract_features(str(image_path))
                image_id = image_path.stem
                self.database_features[image_id] = features
                self.database_paths[image_id] = str(image_path)
            except Exception as e:
                print(f"Warning: Could not process {image_path}: {e}")
        
        print(f"Built index with {len(self.database_features)} images")
    
    def query(self, query_image_path: str, top_k: int = 5) -> List[Tuple[str, float, str]]:
        """
        Query the index with an image and return ranked results.
        
        Args:
            query_image_path: Path to the query image
            top_k: Number of top results to return
            
        Returns:
            List of tuples (image_id, similarity_score, image_path) ranked by similarity
        """
        if not self.database_features:
            raise ValueError("Index is empty. Call build_index() first.")
        
        # Extract features from query image
        query_features = self.feature_extractor.extract_features(query_image_path)
        
        # Calculate similarities with all database images
        database_ids = list(self.database_features.keys())
        database_feature_matrix = np.array([self.database_features[id] for id in database_ids])
        
        similarities = self.similarity_calculator.calculate_batch_similarity(
            query_features, database_feature_matrix
        )
        
        # Rank results by similarity (descending)
        ranked_indices = np.argsort(similarities)[::-1]
        
        # Get top-k results
        results = []
        for i in ranked_indices[:top_k]:
            image_id = database_ids[i]
            similarity_score = float(similarities[i])
            image_path = self.database_paths[image_id]
            results.append((image_id, similarity_score, image_path))
        
        return results
    
    def get_index_size(self) -> int:
        """Get the number of images in the index."""
        return len(self.database_features)
    
    def clear_index(self) -> None:
        """Clear the current index."""
        self.database_features = {}
        self.database_paths = {}

# Baseline Visual Retrieval Method

## Overview
This document describes the baseline visual retrieval method implemented in Phase 3 of the project. The baseline provides a simple, explainable approach to image similarity that can be used for comparison against more advanced deep learning methods.

## Baseline Method

### Feature Representation
The baseline uses color histogram features:
- **Color Space**: RGB color space
- **Bin Configuration**: 32 bins per color channel
- **Normalization**: Histograms are normalized to sum to 1
- **Total Dimension**: 96 features (32 bins × 3 channels)

### Feature Extraction Process
1. **Image Loading**: Load image and convert to RGB format
2. **Resizing**: Resize to consistent dimensions (128×128 pixels)
3. **Histogram Calculation**: Compute histograms for each color channel
4. **Normalization**: Normalize histograms to represent probability distributions
5. **Feature Concatenation**: Combine all channel features into single vector

### Similarity Calculation
The baseline uses Euclidean distance converted to similarity:
- **Distance Metric**: Euclidean distance between feature vectors
- **Similarity Conversion**: `similarity = 1 / (1 + distance)`
- **Range**: Similarity scores range from 0 (completely different) to 1 (identical)

### Retrieval Process
1. **Index Building**: Extract features for all database images
2. **Query Processing**: Extract features from query image
3. **Similarity Computation**: Calculate similarity between query and all database images
4. **Ranking**: Sort images by similarity score (descending)
5. **Result Selection**: Return top-k most similar images

## Limitations

### Current Limitations
1. **Color Sensitivity**: Highly sensitive to lighting conditions and color variations
2. **Spatial Information**: Ignores spatial relationships between image regions
3. **Semantic Understanding**: Cannot capture semantic similarity beyond color distribution
4. **Scale Invariance**: Not invariant to scale changes
5. **Rotation Sensitivity**: Sensitive to image rotation

### Comparison with Deep Learning
The baseline will be compared against deep embedding models (CLIP, SigLIP) on:
- **Retrieval Accuracy**: Precision@k, Recall@k, mAP
- **Robustness**: Performance across different lighting, angles, and conditions
- **Semantic Understanding**: Ability to capture semantic similarity
- **Feature Dimensionality**: Compactness and efficiency of feature representation

The baseline implements clear interfaces that allow for easy replacement:

"""
Minimal classical image retrieval baseline using color histograms.

Provides:
- extract_features(image_path): Extract normalized RGB color histogram
- cosine_similarity(feature_a, feature_b): Compute cosine similarity
- rank_candidates(query_image, candidate_images, top_k=5): Rank candidates
"""

import numpy as np
from PIL import Image
from pathlib import Path


def extract_features(image_path):
    """
    Extract a normalized RGB color histogram feature from an image.
    
    Args:
        image_path: Path to the image file (str or Path)
        
    Returns:
        numpy.ndarray: Normalized feature vector (dtype=float64)
        
    Raises:
        FileNotFoundError: If image_path does not exist
        ValueError: If image cannot be opened or is invalid
    """
    path = Path(image_path)
    if not path.exists():
        raise FileNotFoundError(f"Image not found: {image_path}")
    
    try:
        img = Image.open(path)
        img = img.convert('RGB')
        img = img.resize((128, 128))
    except Exception as e:
        raise ValueError(f"Cannot open image {image_path}: {e}")
    
    arr = np.array(img)
    
    # Compute histograms for each channel
    r_hist, _ = np.histogram(arr[:, :, 0], bins=32, range=(0, 255))
    g_hist, _ = np.histogram(arr[:, :, 1], bins=32, range=(0, 255))
    b_hist, _ = np.histogram(arr[:, :, 2], bins=32, range=(0, 255))
    
    # Concatenate and normalize
    feature = np.concatenate([r_hist, g_hist, b_hist]).astype(np.float64)
    
    norm = np.linalg.norm(feature)
    if norm > 0:
        feature = feature / norm
    
    return feature


def cosine_similarity(feature_a, feature_b):
    """
    Compute cosine similarity between two feature vectors.
    
    Args:
        feature_a: First feature vector (numpy.ndarray)
        feature_b: Second feature vector (numpy.ndarray)
        
    Returns:
        float: Cosine similarity score (range: -1 to 1, but typically 0-1 for histograms)
        
    Raises:
        ValueError: If feature vectors have different dimensions or are zero-length
    """
    if len(feature_a) != len(feature_b):
        raise ValueError(f"Feature dimension mismatch: {len(feature_a)} vs {len(feature_b)}")
    
    if len(feature_a) == 0:
        raise ValueError("Feature vectors cannot be empty")
    
    dot_product = np.dot(feature_a, feature_b)
    norm_a = np.linalg.norm(feature_a)
    norm_b = np.linalg.norm(feature_b)
    
    if norm_a == 0 or norm_b == 0:
        return 0.0
    
    return float(dot_product / (norm_a * norm_b))


def rank_candidates(query_image, candidate_images, top_k=5):
    """
    Rank candidate images by similarity to the query image.
    
    Args:
        query_image: Path to the query image (str or Path)
        candidate_images: List of paths to candidate images
        top_k: Number of top results to return (default: 5)
        
    Returns:
        list of tuples: [(image_path, similarity_score), ...] sorted by score descending
        
    Raises:
        FileNotFoundError: If query_image or any candidate does not exist
        ValueError: If images cannot be processed
    """
    # Extract query features
    query_features = extract_features(query_image)
    
    # Score each candidate
    scored_candidates = []
    for candidate_path in candidate_images:
        try:
            candidate_features = extract_features(candidate_path)
            score = cosine_similarity(query_features, candidate_features)
            scored_candidates.append((str(candidate_path), score))
        except (FileNotFoundError, ValueError) as e:
            # Re-raise with context
            raise ValueError(f"Error processing candidate {candidate_path}: {e}")
    
    # Sort by similarity descending
    scored_candidates.sort(key=lambda x: x[1], reverse=True)
    
    # Return top-k
    return scored_candidates[:top_k]

"""
Deep Visual Embedding Interface

Provides a modular interface for visual embedding models that can be
benchmarked against the RGB histogram baseline. This module defines
the contract for embedding models and includes a placeholder implementation
for testing purposes.

The interface is designed to be replaceable with CLIP, SigLIP, or other
pretrained vision models without changing the retrieval pipeline.
"""

from abc import ABC, abstractmethod
from typing import List, Union, Optional
import numpy as np
from pathlib import Path
import pickle
import hashlib


class VisualEmbeddingModel(ABC):
    """
    Abstract base class for visual embedding models.
    
    All embedding models must implement this interface to ensure
    compatibility with the retrieval pipeline and benchmarking framework.
    """
    
    @abstractmethod
    def load_model(self) -> None:
        """
        Load the embedding model into memory.
        
        This method should be called before generating embeddings.
        """
        pass
    
    @abstractmethod
    def embed_image(self, image_path: Union[str, Path]) -> np.ndarray:
        """
        Generate embedding vector for a single image.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Embedding vector as numpy array
        """
        pass
    
    def embed_images(self, image_paths: List[Union[str, Path]]) -> np.ndarray:
        """
        Generate embeddings for multiple images.
        
        Args:
            image_paths: List of paths to image files
            
        Returns:
            Matrix of embedding vectors (n_images x embedding_dim)
        """
        embeddings = []
        for path in image_paths:
            embeddings.append(self.embed_image(path))
        return np.array(embeddings)
    
    @abstractmethod
    def get_embedding_dimension(self) -> int:
        """
        Get the dimensionality of the embedding vectors.
        
        Returns:
            Embedding vector dimension
        """
        pass
    
    def normalize_embedding(self, embedding: np.ndarray) -> np.ndarray:
        """
        Normalize an embedding vector to unit length.
        
        Args:
            embedding: Input embedding vector
            
        Returns:
            L2-normalized embedding vector
        """
        norm = np.linalg.norm(embedding)
        if norm > 0:
            return embedding / norm
        return embedding
    
    def normalize_embeddings(self, embeddings: np.ndarray) -> np.ndarray:
        """
        Normalize multiple embedding vectors to unit length.
        
        Args:
            embeddings: Matrix of embedding vectors
            
        Returns:
            Matrix of L2-normalized embedding vectors
        """
        norms = np.linalg.norm(embeddings, axis=1, keepdims=True)
        norms[norms == 0] = 1  # Avoid division by zero
        return embeddings / norms
    
    def compute_similarity(self, embedding1: np.ndarray, embedding2: np.ndarray) -> float:
        """
        Compute cosine similarity between two embedding vectors.
        
        Args:
            embedding1: First embedding vector
            embedding2: Second embedding vector
            
        Returns:
            Cosine similarity score
        """
        # Normalize embeddings
        emb1_norm = self.normalize_embedding(embedding1)
        emb2_norm = self.normalize_embedding(embedding2)
        
        # Compute cosine similarity
        dot_product = np.dot(emb1_norm, emb2_norm)
        return float(dot_product)
    
    def compute_batch_similarity(self, query_embedding: np.ndarray, 
                               database_embeddings: np.ndarray) -> np.ndarray:
        """
        Compute cosine similarity between a query and a batch of database embeddings.
        
        Args:
            query_embedding: Query embedding vector
            database_embeddings: Matrix of database embedding vectors
            
        Returns:
            Array of similarity scores
        """
        # Normalize embeddings
        query_norm = self.normalize_embedding(query_embedding)
        database_norms = self.normalize_embeddings(database_embeddings)
        
        # Compute cosine similarities
        similarities = np.dot(database_norms, query_norm)
        return similarities
    
    def cache_embedding(self, image_path: Union[str, Path], 
                       embedding: np.ndarray, cache_dir: Union[str, Path]) -> None:
        """
        Cache an embedding to disk for faster retrieval in subsequent runs.
        
        Args:
            image_path: Path to the original image
            embedding: Embedding vector to cache
            cache_dir: Directory to store cached embeddings
        """
        cache_dir = Path(cache_dir)
        cache_dir.mkdir(parents=True, exist_ok=True)
        
        # Create a unique filename based on image path and hash
        path_hash = hashlib.md5(str(image_path).encode()).hexdigest()
        cache_file = cache_dir / f"{path_hash}.pkl"
        
        with open(cache_file, 'wb') as f:
            pickle.dump(embedding, f)
    
    def load_cached_embedding(self, image_path: Union[str, Path], 
                            cache_dir: Union[str, Path]) -> Optional[np.ndarray]:
        """
        Load a cached embedding from disk.
        
        Args:
            image_path: Path to the original image
            cache_dir: Directory containing cached embeddings
            
        Returns:
            Cached embedding vector or None if not found
        """
        cache_dir = Path(cache_dir)
        if not cache_dir.exists():
            return None
        
        path_hash = hashlib.md5(str(image_path).encode()).hexdigest()
        cache_file = cache_dir / f"{path_hash}.pkl"
        
        if cache_file.exists():
            with open(cache_file, 'rb') as f:
                return pickle.load(f)
        return None


class PlaceholderEmbeddingModel(VisualEmbeddingModel):
    """
    Placeholder embedding model for testing and benchmarking.
    
    This model generates deterministic embeddings using a fixed random
    projection matrix. It is intended for testing the interface and
    will be replaced with a pretrained model (CLIP, SigLIP, etc.) in later phases.
    """
    
    def __init__(self, embedding_dim: int = 512, seed: int = 42):
        """
        Initialize the placeholder embedding model.
        
        Args:
            embedding_dim: Dimensionality of embedding vectors
            seed: Random seed for reproducibility
        """
        self.embedding_dim = embedding_dim
        self.seed = seed
        self._model_loaded = False
        
        # Create a fixed random projection matrix for deterministic embeddings
        np.random.seed(seed)
        # We'll use a simple color-based projection as a placeholder
        # This is not a real deep learning model, just for interface testing
        self.projection_matrix = np.random.randn(3 * 32, embedding_dim)  # 3*32 from color histogram
    
    def load_model(self) -> None:
        """Load the placeholder model (no-op for this implementation)."""
        self._model_loaded = True
    
    def embed_image(self, image_path: Union[str, Path]) -> np.ndarray:
        """
        Generate a deterministic embedding for an image using color histogram projection.
        
        This is a placeholder implementation that creates embeddings from
        color histograms projected to a higher dimensional space.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Embedding vector
        """
        if not self._model_loaded:
            self.load_model()
        
        # Extract color histogram features (same as baseline)
        from PIL import Image
        import numpy as np
        
        try:
            img = Image.open(image_path).convert('RGB')
            img = img.resize((128, 128))
        except Exception as e:
            raise ValueError(f"Cannot open image {image_path}: {e}")
        
        arr = np.array(img)
        
        # Compute histograms for each channel
        r_hist, _ = np.histogram(arr[:, :, 0], bins=32, range=(0, 255))
        g_hist, _ = np.histogram(arr[:, :, 1], bins=32, range=(0, 255))
        b_hist, _ = np.histogram(arr[:, :, 2], bins=32, range=(0, 255))
        
        # Concatenate and normalize
        hist_feature = np.concatenate([r_hist, g_hist, b_hist]).astype(np.float64)
        
        # Project to embedding space using the fixed matrix
        embedding = np.dot(hist_feature, self.projection_matrix)
        
        # Normalize the embedding
        embedding = self.normalize_embedding(embedding)
        
        return embedding
    
    def get_embedding_dimension(self) -> int:
        """Get the dimensionality of the embedding vectors."""
        return self.embedding_dim


class NormalizedEmbeddingModel(VisualEmbeddingModel):
    """
    Wrapper class that ensures embeddings are always normalized.
    
    This can be used to wrap any embedding model to guarantee normalized outputs.
    """
    
    def __init__(self, model: VisualEmbeddingModel):
        """
        Initialize with a model to wrap.
        
        Args:
            model: The embedding model to wrap
        """
        self.model = model
    
    def load_model(self) -> None:
        """Load the wrapped model."""
        self.model.load_model()
    
    def embed_image(self, image_path: Union[str, Path]) -> np.ndarray:
        """Generate a normalized embedding for an image."""
        embedding = self.model.embed_image(image_path)
        return self.model.normalize_embedding(embedding)
    
    def get_embedding_dimension(self) -> int:
        """Get the dimensionality of the embedding vectors."""
        return self.model.get_embedding_dimension()


def create_embedding_model(model_type: str = "placeholder", **kwargs) -> VisualEmbeddingModel:
    """
    Factory function to create embedding models.
    
    Args:
        model_type: Type of embedding model to create
        **kwargs: Additional arguments for the model
        
    Returns:
        VisualEmbeddingModel instance
        
    Raises:
        ValueError: If model_type is not supported
    """
    if model_type == "placeholder":
        return PlaceholderEmbeddingModel(**kwargs)
    elif model_type == "normalized_placeholder":
        model = PlaceholderEmbeddingModel(**kwargs)
        return NormalizedEmbeddingModel(model)
    else:
        raise ValueError(f"Unsupported model type: {model_type}")


def benchmark_against_baseline(query_image: Union[str, Path], 
                             candidate_images: List[Union[str, Path]],
                             embedding_model: VisualEmbeddingModel,
                             top_k: int = 5) -> List[tuple]:
    """
    Benchmark an embedding model against the RGB histogram baseline.
    
    Args:
        query_image: Path to the query image
        candidate_images: List of paths to candidate images
        embedding_model: Embedding model to evaluate
        top_k: Number of top results to return
        
    Returns:
        List of tuples (image_path, similarity_score) ranked by similarity
    """
    # Generate embeddings
    query_embedding = embedding_model.embed_image(query_image)
    candidate_embeddings = embedding_model.embed_images(candidate_images)
    
    # Compute similarities
    similarities = embedding_model.compute_batch_similarity(query_embedding, candidate_embeddings)
    
    # Rank candidates
    scored_candidates = list(zip([str(p) for p in candidate_images], similarities))
    scored_candidates.sort(key=lambda x: x[1], reverse=True)
    
    return scored_candidates[:top_k]

# Deep Visual Embedding Evaluation

## Overview
This document describes the deep visual embedding approach and how it will be evaluated against the RGB histogram baseline.

## Why Deep Visual Embeddings?

### Limitations of RGB Histogram Baseline
The RGB histogram baseline implemented in Phase 3 has several limitations:
- **Color Sensitivity**: Highly sensitive to lighting conditions and color variations
- **Spatial Information**: Ignores spatial relationships between image regions
- **Semantic Understanding**: Cannot capture semantic similarity beyond color distribution
- **Scale Invariance**: Not invariant to scale changes
- **Rotation Sensitivity**: Sensitive to image rotation

### Advantages of Deep Visual Embeddings
Deep visual embeddings address these limitations by:
- **Semantic Understanding**: Capturing high-level semantic features through deep neural networks
- **Robustness**: Being more invariant to lighting, rotation, and scale changes
- **Spatial Awareness**: Incorporating spatial information through convolutional layers
- **Transfer Learning**: Leveraging pretrained models trained on large datasets

## How Deep Embeddings Differ from RGB Histogram Baseline

### Feature Representation
- **RGB Histogram**: 96-dimensional vector (32 bins × 3 color channels)
- **Deep Embedding**: Typically 512-1024 dimensional vectors from pretrained models

### Feature Extraction Process
- **RGB Histogram**: Manual feature engineering with fixed binning
- **Deep Embedding**: Automatic feature learning through neural network layers

### Similarity Calculation
- **RGB Histogram**: Euclidean distance converted to similarity
- **Deep Embedding**: Cosine similarity in embedding space

## Embedding Model Interface

The deep embedding system implements a clean interface defined in `ml/visual_embedding.py`:

```python

    def compute_batch_similarity(self, query: np.ndarray, database: np.ndarray) -> np.ndarray: ...
```

## Evaluation Framework

### Metrics
The embedding models will be evaluated using:
- **Precision@k**: Percentage of relevant images in top-k results
- **Recall@k**: Percentage of relevant images retrieved in top-k
- **Mean Average Precision (mAP)**: Mean of precision at different recall levels
- **Retrieval Speed**: Time to generate embeddings and compute similarities

### Benchmark Dataset
A small, curated benchmark dataset will be created for evaluation:
- Synthetic test images with known similarities
- Real heritage images (if available and permitted)
- Ground truth similarity matrices

### Comparison Methodology
1. Generate embeddings for all images in the benchmark dataset
2. Compute similarity matrices for each model
3. Run retrieval queries and calculate metrics
4. Compare results against the RGB histogram baseline

## Model Selection Criteria

### Performance Metrics
- **Retrieval Accuracy**: Primary criterion (mAP, Precision@k)
- **Invariance**: Robustness to lighting, rotation, scale changes
- **Semantic Understanding**: Ability to capture semantic similarity

### Practical Considerations
- **Inference Speed**: Time to process images
- **Model Size**: Memory footprint and storage requirements
- **Dependency Requirements**: Additional libraries or hardware needs

### Decision Process
Final model selection will be based on:
1. Empirical evaluation results on the benchmark dataset
2. Trade-off analysis between accuracy and practical constraints
3. Compatibility with the existing retrieval pipeline

"""
Deep Visual Embedding Interface

Provides a modular interface for visual embedding models that can be
benchmarked against the RGB histogram baseline. This module defines
the contract for embedding models and includes a placeholder implementation
for testing purposes.

The interface is designed to be replaceable with CLIP, SigLIP, or other
pretrained vision models without changing the retrieval pipeline.
"""

from abc import ABC, abstractmethod
from typing import List, Union, Optional
import numpy as np
from pathlib import Path
import pickle
import hashlib


class VisualEmbeddingModel(ABC):
    """
    Abstract base class for visual embedding models.
    
    All embedding models must implement this interface to ensure
    compatibility with the retrieval pipeline and benchmarking framework.
    """
    
    @abstractmethod
    def load_model(self) -> None:
        """
        Load the embedding model into memory.
        
        This method should be called before generating embeddings.
        """
        pass
    
    @abstractmethod
    def embed_image(self, image_path: Union[str, Path]) -> np.ndarray:
        """
        Generate embedding vector for a single image.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Embedding vector as numpy array
        """
        pass
    
    def embed_images(self, image_paths: List[Union[str, Path]]) -> np.ndarray:
        """
        Generate embeddings for multiple images.
        
        Args:
            image_paths: List of paths to image files
            
        Returns:
            Matrix of embedding vectors (n_images x embedding_dim)
        """
        embeddings = []
        for path in image_paths:
            embeddings.append(self.embed_image(path))
        return np.array(embeddings)
    
    @abstractmethod
    def get_embedding_dimension(self) -> int:
        """
        Get the dimensionality of the embedding vectors.
        
        Returns:
            Embedding vector dimension
        """
        pass
    
    def normalize_embedding(self, embedding: np.ndarray) -> np.ndarray:
        """
        Normalize an embedding vector to unit length.
        
        Args:
            embedding: Input embedding vector
            
        Returns:
            L2-normalized embedding vector
        """
        norm = np.linalg.norm(embedding)
        if norm > 0:
            return embedding / norm
        return embedding
    
    def normalize_embeddings(self, embeddings: np.ndarray) -> np.ndarray:
        """
        Normalize multiple embedding vectors to unit length.
        
        Args:
            embeddings: Matrix of embedding vectors
            
        Returns:
            Matrix of L2-normalized embedding vectors
        """
        norms = np.linalg.norm(embeddings, axis=1, keepdims=True)
        norms[norms == 0] = 1  # Avoid division by zero
        return embeddings / norms
    
    def compute_similarity(self, embedding1: np.ndarray, embedding2: np.ndarray) -> float:
        """
        Compute cosine similarity between two embedding vectors.
        
        Args:
            embedding1: First embedding vector
            embedding2: Second embedding vector
            
        Returns:
            Cosine similarity score
        """
        # Normalize embeddings
        emb1_norm = self.normalize_embedding(embedding1)
        emb2_norm = self.normalize_embedding(embedding2)
        
        # Compute cosine similarity
        dot_product = np.dot(emb1_norm, emb2_norm)
        return float(dot_product)
    
    def compute_batch_similarity(self, query_embedding: np.ndarray, 
                               database_embeddings: np.ndarray) -> np.ndarray:
        """
        Compute cosine similarity between a query and a batch of database embeddings.
        
        Args:
            query_embedding: Query embedding vector
            database_embeddings: Matrix of database embedding vectors
            
        Returns:
            Array of similarity scores
        """
        # Normalize embeddings
        query_norm = self.normalize_embedding(query_embedding)
        database_norms = self.normalize_embeddings(database_embeddings)
        
        # Compute cosine similarities
        similarities = np.dot(database_norms, query_norm)
        return similarities
    
    def cache_embedding(self, image_path: Union[str, Path], 
                       embedding: np.ndarray, cache_dir: Union[str, Path]) -> None:
        """
        Cache an embedding to disk for faster retrieval in subsequent runs.
        
        Args:
            image_path: Path to the original image
            embedding: Embedding vector to cache
            cache_dir: Directory to store cached embeddings
        """
        cache_dir = Path(cache_dir)
        cache_dir.mkdir(parents=True, exist_ok=True)
        
        # Create a unique filename based on image path and hash
        path_hash = hashlib.md5(str(image_path).encode()).hexdigest()
        cache_file = cache_dir / f"{path_hash}.pkl"
        
        with open(cache_file, 'wb') as f:
            pickle.dump(embedding, f)
    
    def load_cached_embedding(self, image_path: Union[str, Path], 
                            cache_dir: Union[str, Path]) -> Optional[np.ndarray]:
        """
        Load a cached embedding from disk.
        
        Args:
            image_path: Path to the original image
            cache_dir: Directory containing cached embeddings
            
        Returns:
            Cached embedding vector or None if not found
        """
        cache_dir = Path(cache_dir)
        if not cache_dir.exists():
            return None
        
        path_hash = hashlib.md5(str(image_path).encode()).hexdigest()
        cache_file = cache_dir / f"{path_hash}.pkl"
        
        if cache_file.exists():
            with open(cache_file, 'rb') as f:
                return pickle.load(f)
        return None


class PlaceholderEmbeddingModel(VisualEmbeddingModel):
    """
    Placeholder embedding model for testing and benchmarking.
    
    This model generates deterministic embeddings using a fixed random
    projection matrix. It is intended for testing the interface and
    will be replaced with a pretrained model (CLIP, SigLIP, etc.) in later phases.
    """
    
    def __init__(self, embedding_dim: int = 512, seed: int = 42):
        """
        Initialize the placeholder embedding model.
        
        Args:
            embedding_dim: Dimensionality of embedding vectors
            seed: Random seed for reproducibility
        """
        self.embedding_dim = embedding_dim
        self.seed = seed
        self._model_loaded = False
        
        # Create a fixed random projection matrix for deterministic embeddings
        np.random.seed(seed)
        # We'll use a simple color-based projection as a placeholder
        # This is not a real deep learning model, just for interface testing
        self.projection_matrix = np.random.randn(3 * 32, embedding_dim)  # 3*32 from color histogram
    
    def load_model(self) -> None:
        """Load the placeholder model (no-op for this implementation)."""
        self._model_loaded = True
    
    def embed_image(self, image_path: Union[str, Path]) -> np.ndarray:
        """
        Generate a deterministic embedding for an image using color histogram projection.
        
        This is a placeholder implementation that creates embeddings from
        color histograms projected to a higher dimensional space.
        
        Args:
            image_path: Path to the image file
            
        Returns:
            Embedding vector
        """
        if not self._model_loaded:
            self.load_model()
        
        # Extract color histogram features (same as baseline)
        from PIL import Image
        import numpy as np
        
        try:
            img = Image.open(image_path).convert('RGB')
            img = img.resize((128, 128))
        except Exception as e:
            raise ValueError(f"Cannot open image {image_path}: {e}")
        
        arr = np.array(img)
        
        # Compute histograms for each channel
        r_hist, _ = np.histogram(arr[:, :, 0], bins=32, range=(0, 255))
        g_hist, _ = np.histogram(arr[:, :, 1], bins=32, range=(0, 255))
        b_hist, _ = np.histogram(arr[:, :, 2], bins=32, range=(0, 255))
        
        # Concatenate and normalize
        hist_feature = np.concatenate([r_hist, g_hist, b_hist]).astype(np.float64)
        
        # Project to embedding space using the fixed matrix
        embedding = np.dot(hist_feature, self.projection_matrix)
        
        # Normalize the embedding
        embedding = self.normalize_embedding(embedding)
        
        return embedding
    
    def get_embedding_dimension(self) -> int:
        """Get the dimensionality of the embedding vectors."""
        return self.embedding_dim


class NormalizedEmbeddingModel(VisualEmbeddingModel):
    """
    Wrapper class that ensures embeddings are always normalized.
    
    This can be used to wrap any embedding model to guarantee normalized outputs.
    """
    
    def __init__(self, model: VisualEmbeddingModel):
        """
        Initialize with a model to wrap.
        
        Args:
            model: The embedding model to wrap
        """
        self.model = model
    
    def load_model(self) -> None:
        """Load the wrapped model."""
        self.model.load_model()
    
    def embed_image(self, image_path: Union[str, Path]) -> np.ndarray:
        """Generate a normalized embedding for an image."""
        embedding = self.model.embed_image(image_path)
        return self.model.normalize_embedding(embedding)
    
    def get_embedding_dimension(self) -> int:
        """Get the dimensionality of the embedding vectors."""
        return self.model.get_embedding_dimension()


def create_embedding_model(model_type: str = "placeholder", **kwargs) -> VisualEmbeddingModel:
    """
    Factory function to create embedding models.
    
    Args:
        model_type: Type of embedding model to create
        **kwargs: Additional arguments for the model
        
    Returns:
        VisualEmbeddingModel instance
        
    Raises:
        ValueError: If model_type is not supported
    """
    if model_type == "placeholder":
        return PlaceholderEmbeddingModel(**kwargs)
    elif model_type == "normalized_placeholder":
        model = PlaceholderEmbeddingModel(**kwargs)
        return NormalizedEmbeddingModel(model)
    else:
        raise ValueError(f"Unsupported model type: {model_type}")


def benchmark_against_baseline(query_image: Union[str, Path], 
                             candidate_images: List[Union[str, Path]],
                             embedding_model: VisualEmbeddingModel,
                             top_k: int = 5) -> List[tuple]:
    """
    Benchmark an embedding model against the RGB histogram baseline.
    
    Args:
        query_image: Path to the query image
        candidate_images: List of paths to candidate images
        embedding_model: Embedding model to evaluate
        top_k: Number of top results to return
        
    Returns:
        List of tuples (image_path, similarity_score) ranked by similarity
    """
    # Generate embeddings
    query_embedding = embedding_model.embed_image(query_image)
    candidate_embeddings = embedding_model.embed_images(candidate_images)
    
    # Compute similarities
    similarities = embedding_model.compute_batch_similarity(query_embedding, candidate_embeddings)
    
    # Rank candidates
    scored_candidates = list(zip([str(p) for p in candidate_images], similarities))
    scored_candidates.sort(key=lambda x: x[1], reverse=True)
    
    return scored_candidates[:top_k]

# Deep Visual Embedding Evaluation

## Overview
This document describes the deep visual embedding approach and how it will be evaluated against the RGB histogram baseline.

## Why Deep Visual Embeddings?

### Limitations of RGB Histogram Baseline
The RGB histogram baseline implemented in Phase 3 has several limitations:
- **Color Sensitivity**: Highly sensitive to lighting conditions and color variations
- **Spatial Information**: Ignores spatial relationships between image regions
- **Semantic Understanding**: Cannot capture semantic similarity beyond color distribution
- **Scale Invariance**: Not invariant to scale changes
- **Rotation Sensitivity**: Sensitive to image rotation

### Advantages of Deep Visual Embeddings
Deep visual embeddings address these limitations by:
- **Semantic Understanding**: Capturing high-level semantic features through deep neural networks
- **Robustness**: Being more invariant to lighting, rotation, and scale changes
- **Spatial Awareness**: Incorporating spatial information through convolutional layers
- **Transfer Learning**: Leveraging pretrained models trained on large datasets

## How Deep Embeddings Differ from RGB Histogram Baseline

### Feature Representation
- **RGB Histogram**: 96-dimensional vector (32 bins × 3 color channels)
- **Deep Embedding**: Typically 512-1024 dimensional vectors from pretrained models

### Feature Extraction Process
- **RGB Histogram**: Manual feature engineering with fixed binning
- **Deep Embedding**: Automatic feature learning through neural network layers

### Similarity Calculation
- **RGB Histogram**: Euclidean distance converted to similarity
- **Deep Embedding**: Cosine similarity in embedding space

## Embedding Model Interface

"""
Minimal classical image retrieval baseline using color histograms.

Provides:
- extract_features(image_path): Extract normalized RGB color histogram
- cosine_similarity(feature_a, feature_b): Compute cosine similarity
- rank_candidates(query_image, candidate_images, top_k=5): Rank candidates
"""

import numpy as np
from PIL import Image
from pathlib import Path


def extract_features(image_path):
    """
    Extract a normalized RGB color histogram feature from an image.
    
    Args:
        image_path: Path to the image file (str or Path)
        
    Returns:
        numpy.ndarray: Normalized feature vector (dtype=float64)
        
    Raises:
        FileNotFoundError: If image_path does not exist
        ValueError: If image cannot be opened or is invalid
    """
    path = Path(image_path)
    if not path.exists():
        raise FileNotFoundError(f"Image not found: {image_path}")
    
    try:
        img = Image.open(path)
        img = img.convert('RGB')
        img = img.resize((128, 128))
    except Exception as e:
        raise ValueError(f"Cannot open image {image_path}: {e}")
    
    arr = np.array(img)
    
    # Compute histograms for each channel
    r_hist, _ = np.histogram(arr[:, :, 0], bins=32, range=(0, 255))
    g_hist, _ = np.histogram(arr[:, :, 1], bins=32, range=(0, 255))
    b_hist, _ = np.histogram(arr[:, :, 2], bins=32, range=(0, 255))
    
    # Concatenate and normalize
    feature = np.concatenate([r_hist, g_hist, b_hist]).astype(np.float64)
    
    norm = np.linalg.norm(feature)
    if norm > 0:
        feature = feature / norm
    
    return feature


def cosine_similarity(feature_a, feature_b):
    """
    Compute cosine similarity between two feature vectors.
    
    Args:
        feature_a: First feature vector (numpy.ndarray)
        feature_b: Second feature vector (numpy.ndarray)
        
    Returns:
        float: Cosine similarity score (range: -1 to 1, but typically 0-1 for histograms)
        
    Raises:
        ValueError: If feature vectors have different dimensions or are zero-length
    """
    if len(feature_a) != len(feature_b):
        raise ValueError(f"Feature dimension mismatch: {len(feature_a)} vs {len(feature_b)}")
    
    if len(feature_a) == 0:
        raise ValueError("Feature vectors cannot be empty")
    
    dot_product = np.dot(feature_a, feature_b)
    norm_a = np.linalg.norm(feature_a)
    norm_b = np.linalg.norm(feature_b)
    
    if norm_a == 0 or norm_b == 0:
        return 0.0
    
    return float(dot_product / (norm_a * norm_b))


def rank_candidates(query_image, candidate_images, top_k=5):
    """
    Rank candidate images by similarity to the query image.
    
    Args:
        query_image: Path to the query image (str or Path)
        candidate_images: List of paths to candidate images
        top_k: Number of top results to return (default: 5)
        
    Returns:
        list of tuples: [(image_path, similarity_score), ...] sorted by score descending
        
    Raises:
        FileNotFoundError: If query_image or any candidate does not exist
        ValueError: If images cannot be processed
    """
    # Extract query features
    query_features = extract_features(query_image)
    
    # Score each candidate
    scored_candidates = []
    for candidate_path in candidate_images:
        try:
            candidate_features = extract_features(candidate_path)
            score = cosine_similarity(query_features, candidate_features)
            scored_candidates.append((str(candidate_path), score))
        except (FileNotFoundError, ValueError) as e:
            # Re-raise with context
            raise ValueError(f"Error processing candidate {candidate_path}: {e}")
    
    # Sort by similarity descending
    scored_candidates.sort(key=lambda x: x[1], reverse=True)
    
    # Return top-k
    return scored_candidates[:top_k]
